# GPU / NVIDIA stack for REAL mode. CUDA itself is already installed in the
# runtime; these are the Python components the engine and exporter import:
#   - cudf / cuml / cupy  (RAPIDS): GPU tokenizer dataframes, UMAP + PCA
#   - torch + transformers: the decoder foundation model + tokenizer
#   - xgboost + scikit-learn: the three fraud heads and the preprocessing/PCA
# Installed by scripts/install_deps.py alongside requirements-demo.txt.
#
# RAPIDS and some CUDA wheels live on NVIDIA's index, not PyPI; torch's CUDA 12
# build lives on PyTorch's own index. (Unqualified `torch` on PyPI now pulls a
# CUDA 13 build, which mismatches the cu12 RAPIDS stack and the CUDA 12 driver.)
--extra-index-url https://pypi.nvidia.com
--extra-index-url https://download.pytorch.org/whl/cu121

# RAPIDS — cu12 wheels bundle their own CUDA 12 runtime and only need a CUDA 12
# driver, which the runtime already provides. Keep cudf and cuml versions in
# lockstep; bump both together if the runtime's CUDA driver requires it.
cudf-cu12==24.12.*
cuml-cu12==24.12.*
cupy-cuda12x

# RAPIDS' numba / CUDA-binding companions, pinned to cudf 24.12's tested matrix
# (cudf-cu12 24.12 declares numba-cuda>=0.0.13,<0.0.18 and cuda-python<=12.6).
# numba.cuda is redirected to the numba-cuda package via a .pth hook, so when a
# later unpinned install drifts these forward, EVERY cuDF host copy
# (.to_numpy()/.values) segfaults inside the numba CUDA driver shim — seen live
# as SIGSEGV in export.py _labels via column.data_array_view. Explicit pins make
# install re-runs converge instead of drifting.
numba-cuda==0.0.17
numba==0.60.*
cuda-python==12.6.*

# Decoder foundation model + tokenizer. The +cu121 local version forces the
# CUDA 12 wheel from the PyTorch index (and so can't be satisfied by a PyPI
# cu13 build). cu121 wheels run on any CUDA 12.x driver; cp312 wheels exist.
torch==2.5.1+cu121
# transformers is pinned to the version the checkpoint was saved with
# (config.json "transformers_version": 4.53.3). Unpinned, pip pulls the newest
# release, whose Llama modeling module no longer imports against torch 2.5.1 —
# AutoModelForCausalLM.from_pretrained then dies with "Could not import module
# 'LlamaForCausalLM'. Are this object's requirements defined correctly?".
# Keep these three in lockstep: bump transformers/accelerate only with torch.
transformers==4.53.*
accelerate<2
safetensors
sentencepiece

# Fraud heads + the sklearn preprocessing/PCA used at export time.
xgboost
scikit-learn
numpy
pandas
